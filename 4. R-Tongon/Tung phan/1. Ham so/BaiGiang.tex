\vspace{-0.25cm}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Bậc 1 ($ax+b$):} Phải cùng, trái trái.
        \item \textbf{Bậc 2 ($ax^2+bx+c$):} Trong trái, ngoài cùng.
        \item \textbf{Nhiễm dấu:} Bội lẻ $\to$ đan dấu; Bội chẵn $\to$ không đổi dấu. Khoảng ngoài cùng bên phải cùng dấu với hệ số của mũ lớn nhất.
        \item \textbf{Đạo hàm nhanh:}
        \[
            \left(\frac{ax+b}{cx+d}\right)' = \frac{ad - bc}{(cx+d)^2}; \qquad \left(\frac{ax^2+bx+c}{dx+e}\right)' = \frac{ad \cdot x^2 + 2ae \cdot x + (be - cd)}{(dx+e)^2}
        \]
    \end{itemize}
\end{theorybox}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{2pt}
        \item $f(x)$: xét \textbf{giá trị}.
        \item $f'(x)$: xét \textbf{ĐB, NB, Cực trị}.
        \item \textbf{Kết luận ĐB, NB:} 
        Dùng từ \textbf{``và''} hoặc dấu \textbf{chấm phẩy (;)}. \textcolor{accentred}{\textbf{CẤM}} dùng $\cup$ hoặc $\backslash$.
        \item \textbf{Không thể ĐB, NB trên $\mathbb{R}$:} Lượng giác, hàm có ĐK (mẫu, căn), hàm bậc chẵn.
        \item \textbf{Mũ \& Logarit:}
        \begin{itemize}
            \item $a > 1 \implies$ \textbf{ĐB} (VD: $2^x, \; \log_2 x$).
            \item $0 < a < 1 \implies$ \textbf{NB} (VD: $0{,}5^x, \; \log_{0{,}9} x$).
        \end{itemize}
    \end{itemize}
\end{theorybox}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Cực trị hàm số:} Cực trị là \textbf{nghiệm bội lẻ} của $f'(x) = 0$ và \textbf{phải xác định trên $f(x)$}.
        \item \textbf{Phân biệt thuật ngữ:}\vspace{2pt}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Hàm số đạt cực trị tại $x = \dots$ (\textit{Điểm cực trị của hàm số})
            \item Giá trị cực trị $y = \dots$ (\textit{Cực trị / Cực đại / Cực tiểu})
            \item Điểm cực trị của đồ thị hàm số là $(x; y) = \dots$
        \end{itemize}
        \item \textbf{Bản chất LN, NN:} Là giá trị chặn trên/chặn dưới \textbf{phải tồn tại dấu bằng} tại một điểm thuộc miền xác định.
    \end{itemize}
\end{theorybox}

% 2. BẢN CHẤT XÉT DẤU a, b, c, d
\begin{theorybox}
    \textbf{Xét dấu $a, b, c, d$ của hàm số $y = ax^3 + bx^2 + cx + d$:}
    \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Dấu $a$:} Nhìn nhánh cuối bên phải: \textbf{đi lên} $\implies a > 0$; \textbf{đi xuống} $\implies a < 0$.
        \item \textbf{Dấu $d$:} Đồ thị cắt trục $Oy$ tại điểm $(0; d)$: \textbf{nằm trên $Ox$} $\implies d > 0$; \textbf{nằm dưới $Ox$} $\implies d < 0$.
        \item \textbf{Dấu $b, c$:} Dùng 2 hoành độ cực trị $x_1, x_2$ theo định lý Vi-ét của phương trình $y' = 0$:
        \[
            x_1 + x_2 = -\frac{2b}{3a} \qquad \text{và} \qquad x_1 \cdot x_2 = \frac{c}{3a}
        \]
    \end{itemize}
\end{theorybox}

\vspace{0.15cm}

% 3. CÁC CÁCH TÌM PHƯƠNG TRÌNH BẬC BA & ĐƯỜNG THẲNG ĐI QUA 2 ĐIỂM CỰC TRỊ
\dangbox{KỸ THUẬT}{Tìm phương trình hàm số \& Viết ĐT qua 2 cực trị}

\vspace{0.15cm}

\begin{theorybox}
    \textbf{1. Các cách tìm hàm số bậc ba $y = ax^3 + bx^2 + cx + d$:}
    \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Cách 1 (Lập hệ 4 phương trình):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Dùng toạ độ các điểm đồ thị đi qua.
            \item Dùng toạ độ các điểm cực trị ($y'(x_0) = 0$).
        \end{itemize}
        \item \textbf{Cách 2 (Dùng tương giao):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Cắt trục $Ox$ tại 3 điểm phân biệt: $f(x) = a(x - x_1)(x - x_2)(x - x_3)$.
            \item Tiếp xúc $Ox$ tại $x_1$ và cắt tại $x_2$: $f(x) = a(x - x_1)^2(x - x_2)$.
        \end{itemize}
    \end{itemize}

    \vspace{0.1cm}
    \textbf{2. Phương trình đường thẳng đi qua 2 điểm Cực trị:}
    \begin{itemize}\setlength{\itemsep}{3pt}
        \item Đường thẳng đi qua 2 điểm cực trị chính là \textbf{phần dư của phép chia} $\dfrac{y}{y'}$:
        \[
            y = y' \cdot Q(x) + \underbrace{(Ax + B)}_{\text{phần dư}} \implies \text{Phương trình ĐT là: } y = Ax + B.
        \]
    \end{itemize}
\end{theorybox}

\dangbox{HÀM HỮU TỶ}{1. Hàm bậc nhất / bậc nhất: $y = \dfrac{ax+b}{cx+d}$}

\vspace{0.15cm}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Đạo hàm:} $y' = \dfrac{ad - bc}{(cx + d)^2}$
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Đồng biến} trên từng khoảng xác định $\iff ad - bc > 0$.
            \item \textbf{Nghịch biến} trên từng khoảng xác định $\iff ad - bc < 0$.
        \end{itemize}
        \item \textbf{Cực trị:} Hàm số \textcolor{accentred}{\textbf{KHÔNG CÓ CỰC TRỊ}}.
        \item \textbf{Tiệm cận:}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Tiệm cận đứng (\textbf{TCĐ}): $x = -\dfrac{d}{c}$ (nghiệm của mẫu).
            \item Tiệm cận ngang (\textbf{TCN}): $y = \dfrac{a}{c}$ (hệ số $x$ tử chia mẫu).
        \end{itemize}
        \item \textbf{Cách tìm các hệ số $a, b, c, d$:}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Dùng đường tiệm cận đứng (\textbf{TCĐ}) và tiệm cận ngang (\textbf{TCN}).
            \item Dùng \textbf{điểm đi qua} cụ thể trên đồ thị.
            \item Dùng \textbf{giao điểm} với trục tung $Oy$ ($x=0 \implies y=\frac{b}{d}$) và trục hoành $Ox$ ($y=0 \implies x=-\frac{b}{a}$).
        \end{itemize}
    \end{itemize}
\end{theorybox}

\vspace{0.25cm}

\dangbox{HÀM HỮU TỶ}{2. Hàm bậc hai / bậc nhất: $y = \dfrac{ax^2+bx+c}{dx+e}$}

\vspace{0.15cm}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Tách phần nguyên (chia đa thức):}
        \[
            y = \frac{ax^2+bx+c}{dx+e} = Ax + B + \frac{C}{dx+e}
        \]
        \item \textbf{Số điểm cực trị:} Hàm số có \textbf{0} hoặc \textbf{2 điểm cực trị}.
        \item \textbf{Tiệm cận:}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Tiệm cận đứng (\textbf{TCĐ}): $x = -\dfrac{e}{d}$ (nghiệm của mẫu).
            \item Tiệm cận xiên (\textbf{TCX}): $y = Ax + B$ (phần thương sau khi chia).
        \end{itemize}
        \item \textbf{Phương trình đường thẳng đi qua 2 điểm Cực trị:}
        \[
            y = \frac{(\text{tử})'}{(\text{mẫu})'} = \frac{2ax + b}{d}
        \]
    \end{itemize}
\end{theorybox}

\dangbox{TIỆM CẬN}{Quy tắc nhận diện \& Kỹ thuật bấm máy tính (CALC)}

\vspace{0.2cm}

\begin{theorybox}
    \begin{itemize}\setlength{\itemsep}{3pt}
        \item \textbf{Tiệm cận đứng (TCĐ: $x = x_0$):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Bản chất: Gần như tìm \textbf{Điều kiện (TXĐ)} $\to$ Cho mẫu bằng $0$.
            \item \textbf{Đối với hàm phân số:} Sau khi tìm nghiệm ở mẫu, thay thử lên tử:
            \[
                \text{Thế lên tử} \longrightarrow \begin{cases} \neq 0 \implies \textbf{LẤY} \\ = 0 \text{ hoặc không xác định} \implies \textcolor{accentred}{\textbf{KHÔNG LẤY}} \end{cases}
            \]
        \end{itemize}

        \item \textbf{Tiệm cận ngang (TCN: $y = y_0$):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Bấm máy tính: \textbf{CALC} cho $x \to \pm\infty$ (nhập $10^{10}$ và $-10^{10}$).
        \end{itemize}

        \item \textbf{Tiệm cận xiên (TCX: $y = ax + b$):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item Với những hàm khó mà đề bài yêu cầu tìm TCX $\to$ \textbf{Bấm máy tính}.
            \item \textbf{Bắt buộc} phải CALC kiểm tra cả \textcolor{accentred}{$+\infty$} và \textcolor{accentred}{$-\infty$}:
            \[
                a = \lim_{x \to \pm\infty} \frac{y}{x} \qquad \text{và} \qquad b = \lim_{x \to \pm\infty} (y - ax)
            \]
        \end{itemize}

        \item \textbf{Đọc tiệm cận từ Bảng biến thiên (BBT):}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Khác bản chất là Tiệm cận:} Phải có sự kết hợp giữa \textbf{VÔ CÙNG} ($\pm\infty$) và \textbf{SỐ}:
            \[
                \begin{cases} 
                    x \to x_0 \quad (\text{SỐ}) \implies y \to \pm\infty \quad (\textbf{VÔ CÙNG}) \implies \text{\textbf{TCĐ: }} x = x_0 \\
                    x \to \pm\infty \quad (\textbf{VÔ CÙNG}) \implies y \to y_0 \quad (\text{SỐ}) \implies \text{\textbf{TCN: }} y = y_0
                \end{cases}
            \]
        \end{itemize}

        \item \textbf{Tâm đối xứng \& Tính chẵn -- lẻ:}
        \begin{itemize}\setlength{\itemsep}{2pt}
            \item \textbf{Tâm đối xứng (TĐX):} Là \textbf{giao điểm} của 2 đường tiệm cận.
            \item \textbf{Hàm số lẻ:} $f(-x) = -f(x) \iff f(x) = -f(-x) \implies$ Nhận gốc toạ độ \textbf{$O(0;0)$ làm tâm đối xứng}.
            \item \textbf{Hàm số chẵn:} $f(-x) = f(x) \implies$ Nhận \textbf{trục tung $Oy$ làm trục đối xứng}.
        \end{itemize}
    \end{itemize}
\end{theorybox}
