\dangbox{DẠNG 2}{Khảo sát tính đơn điệu và cực trị từ đồ thị $y = f(x)$}

\questioncore{2}{Cho hàm số bậc ba $y = f(x)$ có đồ thị như hình vẽ bên dưới:
\begin{center}
\begin{tikzpicture}[scale=0.9, >=stealth]
    \draw[->, line width=0.8pt] (-3,0) -- (3,0) node[below] {$x$};
    \draw[->, line width=0.8pt] (0,-4) -- (0,3) node[left] {$y$};
    \node[below left] at (0,0) {$O$};
    
    \draw[dashed] (2,0) -- (2,-4) -- (0,-4);
    \node[below] at (2,0) {$2$};
    \node[left] at (0,-4) {$-4$};
    
    \draw[domain=-1.1:3.1, smooth, variable=\x, line width=1.2pt, deepblue] 
        plot ({\x}, {(\x)^3 - 3*(\x)^2});
    \node[right, deepblue] at (2.5,-1) {$y = f(x)$};
\end{tikzpicture}
\end{center}
Dựa vào đồ thị, hãy:
\begin{enumerate}
    \item Xác định các khoảng đồng biến và nghịch biến của hàm số $y = f(x)$.
    \item Xác định tọa độ điểm cực đại và tọa độ điểm cực tiểu của đồ thị hàm số.
\end{enumerate}
}

\dangbox{DẠNG 3}{Khảo sát tính đơn điệu và cực trị từ đồ thị $y = f'(x)$}

\questioncore{3}{Cho hàm số $y = f(x)$ có đạo hàm $f'(x)$ liên tục trên $\mathbb{R}$. Đồ thị của hàm số đạo hàm $y = f'(x)$ được cho như hình vẽ bên dưới:
\begin{center}
\begin{tikzpicture}[scale=0.85, >=stealth]
    \draw[->, line width=0.8pt] (-3.5,0) -- (3.5,0) node[below] {$x$};
    \draw[->, line width=0.8pt] (0,-3.5) -- (0,3.5) node[left] {$y$};
    \node[below left] at (0,0) {$O$};
    
    \node[below left] at (-2,0) {$-2$};
    \node[below right] at (1,0) {$1$};
    \node[below right] at (3,0) {$3$};
    
    \draw[domain=-2.4:3.4, smooth, variable=\x, line width=1.2pt, accentred] 
        plot ({\x}, {0.5*((\x)+2)*((\x)-1)*((\x)-3)});
    \node[above left, accentred] at (3.3,2.5) {$y = f'(x)$};
\end{tikzpicture}
\end{center}
Dựa vào đồ thị của $y = f'(x)$, hãy:
\begin{enumerate}
    \item Xác định các khoảng đồng biến và nghịch biến của hàm số $y = f(x)$.
    \item Tìm các điểm cực đại và điểm cực tiểu của hàm số $y = f(x)$.
\end{enumerate}
}

\questioncore{9}{Cho hàm số bậc nhất trên bậc nhất $y = \dfrac{ax+b}{cx+d}$ có đồ thị như hình vẽ:
\begin{center}
\begin{tikzpicture}[scale=0.85, >=stealth]
    \clip (-3.5,-3.5) rectangle (4.5,4.5);
    
    \draw[->, line width=0.8pt] (-3.2,0) -- (4,0) node[below] {$x$};
    \draw[->, line width=0.8pt] (0,-3.2) -- (0,4) node[left] {$y$};
    \node[below left] at (0,0) {$O$};
    
    \draw[dashed, line width=0.9pt, accentred] (1,-3.5) -- (1,4.5) node[above right] {$x=1$};
    \draw[dashed, line width=0.9pt, accentred] (-3.5,1) -- (4.5,1) node[above left] {$y=1$};
    
    \draw[domain=-3.2:0.65, smooth, variable=\x, line width=1.2pt, deepblue] 
        plot ({\x}, {((\x)+1)/((\x)-1)});

        plot ({\x}, {((\x)+1)/((\x)-1)});

\end{enumerate}
}

\dangbox{DẠNG 7}{Ứng dụng đạo hàm giải quyết bài toán thực tế}

\questioncore{11}{Một xưởng sản xuất đang gia nhiệt cho một thanh hợp kim. Nhiệt độ $T$ (đơn vị: $^\circ\text{C}$) của thanh hợp kim sau $t$ phút nung ($0 \le t \le 10$) được xác định bởi hàm số:
$$T(t) = -\frac{1}{15}t^3 + t^2 + 16t + 25$$
Hỏi tốc độ tăng nhiệt độ của thanh hợp kim đạt giá trị lớn nhất tại thời điểm nào?}

\questioncore{12}{Một hộ gia đình dự định dùng $120\text{ m}$ lưới rào thép để vây quanh một mảnh đất hình chữ nhật sát một bức tường rào đá có sẵn thành 2 khoang chứa nông sản bằng nhau (như hình vẽ, cạnh áp sát tường đá không cần rào):
\begin{center}
\begin{tikzpicture}[scale=0.85, >=stealth]
    \draw[line width=2.5pt, gray!90] (-0.5,0) -- (6.5,0) node[right, black] {Tường đá};
    \draw[line width=1.2pt, deepblue] (0,0) -- (0,2.5) -- (6,2.5) -- (6,0);
    \draw[line width=1.2pt, deepblue] (3,0) -- (3,2.5);

    \node at (4.5,1.25) {Khoang 2};

    \draw[<->] (0,2.8) -- (6,2.8) node[midway, above] {$y$};
\end{tikzpicture}
\end{center}
Hỏi diện tích lớn nhất mà hộ gia đình có thể rào được là bao nhiêu mét vuông?}

\questioncore{13}{Một công ty sản xuất đồ gia dụng ước tính nếu sản xuất và bán ra $x$ nghìn sản phẩm mỗi tháng thì doanh thu đạt được là $R(x) = -0{,}04x^2 + 500x$ (triệu đồng). Chi phí sản xuất trung bình cho mỗi nghìn sản phẩm được mô hình bởi:
$$C_{tb}(x) = \frac{18000}{x} + 180 \quad (\text{triệu đồng})$$
Hỏi công ty cần sản xuất bao nhiêu nghìn sản phẩm mỗi tháng để thu về lợi nhuận tối đa?}

\questioncore{14}{Người ta cần lắp đặt đường dây cáp viễn thông từ trạm điều khiển đất liền $A$ đến một trạm thu phát trên hải đảo $C$. Khoảng cách ngắn nhất từ đảo $C$ vào bờ biển tại $B$ là $BC = 15\text{ km}$, khoảng cách dọc bờ biển là $BA = 36\text{ km}$. Chi phí đặt cáp ngầm dưới biển từ $C$ đến trạm tiếp sóng $S$ ($S$ nằm giữa $B$ và $A$) là $500$ triệu đồng/km, chi phí rải cáp trên đất liền từ $S$ về $A$ là $300$ triệu đồng/km:
